800:     Structure* structure = baseCell->structure();

812:     }

821: 
822:     if (!conditions.isValid())
823:         return;

825:     PropertyOffset offset = invalidOffset;
826:     CodeBlock::StructureWatchpointMap& watchpointMap = codeBlock->llintGetByIdWatchpointMap();
827:     FixedVector<LLIntPrototypeLoadAdaptiveStructureWatchpoint> watchpoints(conditions.size());
828:     unsigned index = 0;
829:     for (ObjectPropertyCondition condition : conditions) {
830:         auto& watchpoint = watchpoints[index++];
831:         if (!condition.isWatchable(PropertyCondition::MakeNoChanges))
832:             return;
833:         if (condition.condition().kind() == PropertyCondition::Presence)
834:             offset = condition.condition().offset();

837:     }
