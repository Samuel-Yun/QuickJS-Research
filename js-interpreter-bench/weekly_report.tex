\begin{frame}[t]{计时边界收窄后，结果发生反转}
\bodyfont
$r_i=\operatorname{median}(t_{\mathrm{V8},i})/\operatorname{median}(t_{\mathrm{QJS},i})$；
$\mathrm{GM}=\exp\!\left(26^{-1}\sum_{i=1}^{26}\ln r_i\right)$。
\hfill $r>1$：QJS 较低；$r<1$：V8 较低。\par
\vspace{.7mm}
{\centering
\setlength{\tabcolsep}{9pt}\renewcommand{\arraystretch}{1.14}
\begin{tabular}{@{}lrrr@{}}
  \toprule
  Measurement boundary & QuickJS 较低 & V8 较低 & V8/QJS GM\\
  \midrule
  Source-to-finish & 24/26 & 2/26 & 1.694\\
  External amortized & 10/26 & 16/26 & 0.608\\
  Interpreter-mode execution & 5/26 & \blue{21/26} & \blue{0.544}\\
  \bottomrule
\end{tabular}
\par}
\vspace{.5mm}
{\centering
