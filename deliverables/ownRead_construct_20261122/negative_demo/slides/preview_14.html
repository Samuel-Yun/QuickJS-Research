{

return JS_VALUE_GET_TAG(c)==JS_TAG_OBJECT &&

JS_VALUE_GET_TAG(s->negative)==JS_TAG_OBJECT &&

);

static int four_ctor_slow_eligible(JSContext*ctx,FourSite

*s,JSValueConst c,JSValueConst t,unsigned n)

{

if(sparse_same(s,c)){FOLLOW(2);

return 0;}

static void sparse_remember(JSContext*ctx,FourSite*s,

JSValueConst c)

{

if(sparse_same(s,c) ||

JS_VALUE_GET_TAG(c)!=JS_TAG_OBJECT)return;

JSObject*f=JS_VALUE_GET_OBJ(c);

FourRecipe r;

if(f->class_id==JS_CLASS_COLD_BYTECODE_FUNCTION)

return;

if(f->class_id==JS_CLASS_BYTECODE_FUNCTION &&

plan_recipe(f->u.func.function_bytecode,&r))

return;

JSValue old=s->negative;

s->negative=JS_DupValue(ctx,c);

FOLLOW(3);

JS_FreeValue(ctx,old);

}
